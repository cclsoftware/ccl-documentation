struct CCL::ParameterDefinition

Overview

Parameter definition. More…

#include <paramdefinition.h>

struct ParameterDefinition
{
    // fields

    ParameterType type;
    MutableCString className;
    int flags;
    MutableCString name;
    String title;
    int tag;
    Variant minValue;
    Variant maxValue;
    Variant midValue;
    Variant defaultValue;
    Variant precision;
    Variant data;
    MutableCString curveName;
    MutableCString unitName;

    // methods

    bool hasRange () const;
    ParameterDefinition& setType (ParameterType _type);
    ParameterDefinition& setFlags (int _flags);
    ParameterDefinition& setName (StringID _name);
    ParameterDefinition& setTitle (StringRef _title);
    ParameterDefinition& setTag (int _tag);
    ParameterDefinition& setRange (VariantRef _min, VariantRef _max);
    ParameterDefinition& setMid (VariantRef _mid);
    ParameterDefinition& setDefault (VariantRef _default);
    ParameterDefinition& setPrecision (VariantRef _precision);
    ParameterDefinition& setListValue (const ConstVector <String>& strings);
    ParameterDefinition& setListValue (const ConstVector <CStringPtr>& strings);
    ParameterDefinition& setCurve (StringID _curve);
    ParameterDefinition& setUnit (StringID _unit);
};
